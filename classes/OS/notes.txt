  <script>
    // DOM refs
    const initialX = document.getElementById('initialX');
    const currentX = document.getElementById('currentX');
    const delayMs = document.getElementById('delayMs');
    const logEl = document.getElementById('log');
    const btnSequential = document.getElementById('btnSequential');
    const btnInterleaved = document.getElementById('btnInterleaved');
    const btnLocked = document.getElementById('btnLocked');
    const btnClear = document.getElementById('btnClear');
    const btnStep = document.getElementById('btnStep');

    // state
    let X = Number(initialX.value);
    let logCounter = 0;
    let stepMode = false;
    let stepQueue = [];

    function setX(v) {
      X = v;
      currentX.textContent = X;
    }

    function log(message, kind = 'info') {
      const entry = document.createElement('div');
      entry.className = 'log-entry ' + (kind === 'lost' ? 'lost' : (kind === 'ok' ? 'ok' : ''));
      const time = new Date().toLocaleTimeString();
      entry.innerHTML = '<span class="time">[' + time + ']</span>' + message;
      logEl.appendChild(entry);
      logEl.scrollTop = logEl.scrollHeight;
      logCounter++;
    }

    function clearLog() {
      logEl.innerHTML = '<div style="opacity:.8; margin-bottom:6px;">Log (each entry shows which process did which micro-step)</div>';
      logCounter = 0;
    }

    function delay(ms) {
      return new Promise(res => setTimeout(res, ms));
    }

    // Basic micro-operations
    async function loadX(pid) {
      log(`${pid}: Load X (${X})`);
      return X;
    }
    async function addOne(pid, val) {
      const newVal = val + 1;
      log(`${pid}: Add 1 → (${newVal})`);
      return newVal;
    }
    async function storeX(pid, val) {
      const before = X;
      X = val;
      setX(X);
      log(`${pid}: Store X (${val}) -- previous X was ${before}`);
    }

    // Run sequential: P1 full then P2 full
    async function runSequential() {
      setX(Number(initialX.value));
      clearLog();
      const ms = Number(delayMs.value);
      log('--- Sequential run start ---', 'ok');
      // P1
      let v1 = await loadX('P1');
      await delay(ms);
      v1 = await addOne('P1', v1);
      await delay(ms);
      await storeX('P1', v1);
      await delay(ms);
      // P2
      let v2 = await loadX('P2');
      await delay(ms);
      v2 = await addOne('P2', v2);
      await delay(ms);
      await storeX('P2', v2);
      log('--- Sequential run end ---', 'ok');
    }

    // Interleaved naive run (race)
    async function runInterleaved(useLock = false) {
      setX(Number(initialX.value));
      clearLog();
      const ms = Number(delayMs.value);
      log(`--- Interleaved run start (lock: ${useLock ? 'ON' : 'OFF'}) ---`, 'info');

      // Simple lock implementation
      let locked = false;
      const lock = async (pid) => {
        while (locked) {
          log(`${pid}: waiting for lock...`);
          await delay(ms);
        }
        locked = true;
        log(`${pid}: acquired lock`);
      };
      const unlock = (pid) => {
        locked = false;
        log(`${pid}: released lock`);
      };

      // Each process executes the 3 micro-steps with delays; they run concurrently
      const processRoutine = async (pid) => {
        if (useLock) {
          await lock(pid);
        }
        let r = await loadX(pid);
        await delay(ms);
        r = await addOne(pid, r);
        await delay(ms);
        await storeX(pid, r);
        if (useLock) unlock(pid);
      };

      // Start P1 and P2 concurrently (no await)
      const p1 = processRoutine('P1');
      const p2 = processRoutine('P2');

      // wait both
      await Promise.all([p1, p2]);
      log(`--- Interleaved run end; final X = ${X} ---`, 'info');

      // detect lost update: final should be initial + 2 if both increments applied
      const initial = Number(initialX.value);
      if (X === initial + 2) {
        log('Result: both increments applied (no lost update).', 'ok');
      } else {
        log('Result: update lost! final value is less than expected.', 'lost');
      }
    }

    // Step-through mode (execute micro-ops one by one, enqueued)
    function enableStepMode() {
      stepMode = !stepMode;
      btnStep.textContent = stepMode ? 'Exit Step-through' : 'Step-through';
      if (!stepMode) {
        stepQueue = [];
        btnStep.classList.remove('active');
      }
    }

    // Step-run: enqueue actions to be executed manually by clicking the next-step button
    async function runStepThrough() {
      setX(Number(initialX.value));
      clearLog();
      log('--- Step-through mode: you will advance each micro-step manually ---', 'info');

      // Build micro-steps for P1 and P2 interleaved as typical race example
      // For demonstration, we create the sequence:
      // P1: load, P1:add, (pause), P1:store, P2:load, P2:add, P2:store
      stepQueue = [
        async () => { const v = await loadX('P1'); session.lastP1Load = v; },
        async () => { session.lastP1Add = (session.lastP1Load + 1); log(`P1: Add 1 → (${session.lastP1Add})`); },
        async () => { await delay(0); log('--- yield to P2 before P1 store (simulating preemption) ---'); },
        async () => { await storeX('P1', session.lastP1Add); },
        async () => { const v = await loadX('P2'); session.lastP2Load = v; },
        async () => { session.lastP2Add = (session.lastP2Load + 1); log(`P2: Add 1 → (${session.lastP2Add})`); },
        async () => { await storeX('P2', session.lastP2Add); },
        async () => {
          const initial = Number(initialX.value);
          if (X === initial + 2) log('Result: both increments applied (no lost update).', 'ok');
          else log('Result: update lost! final value is less than expected.', 'lost');
        }
      ];

      // store session container for values
      window.session = {};
    }

    // wire buttons
    btnSequential.addEventListener('click', async () => {
      await runSequential();
    });

    btnInterleaved.addEventListener('click', async () => {
      await runInterleaved(false);
    });

    btnLocked.addEventListener('click', async () => {
      await runInterleaved(true);
    });

    btnClear.addEventListener('click', () => clearLog());

    btnStep.addEventListener('click', async () => {
      if (!stepMode) {
        enableStepMode();
        await runStepThrough();
        // show "Next step" button
        const nextBtn = document.createElement('button');
        nextBtn.textContent = 'Next micro-step';
        nextBtn.style.marginLeft = '8px';
        nextBtn.id = 'nextStepBtn';
        document.querySelector('.controls .card:last-child .flex').appendChild(nextBtn);
        nextBtn.addEventListener('click', async () => {
          if (stepQueue.length === 0) {
            log('No more steps. Exit step-through.', 'info');
            nextBtn.remove();
            enableStepMode();
            return;
          }
          const fn = stepQueue.shift();
          await fn();
        });
      } else {
        // disable step mode
        const next = document.getElementById('nextStepBtn');
        if (next) next.remove();
        enableStepMode();
      }
    });

    // Keep display in sync if user edits initial value
    initialX.addEventListener('change', () => {
      setX(Number(initialX.value));
    });

    // init
    setX(Number(initialX.value));
    clearLog();
    log('Ready. Set initial X and delays, then click a run button to simulate.', 'info');
  </script>
